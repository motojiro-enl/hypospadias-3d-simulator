      for (let i = 0; i <= SHAFT_STEPS; i++) {

        const t = (i / SHAFT_STEPS) * 0.88;

        const centerPt = centerShaftCurve.getPointAt(t);

        const tan = centerShaftCurve.getTangentAt(t).normalize();

        const ventralNormal = new THREE.Vector3(0, -tan.z, tan.y).normalize();

        const ventralPt = centerPt.clone().add(ventralNormal.multiplyScalar(ventralSurfaceOffset));

        externalCoursePts.push(ventralPt);

      }


      const subCoronalCenter = centerShaftCurve.getPointAt(0.94);

      const subCoronalTan = centerShaftCurve.getTangentAt(0.94).normalize();

      const subCoronalVentNorm = new THREE.Vector3(0, -subCoronalTan.z, subCoronalTan.y).normalize();

      const subCoronalSurfacePt = subCoronalCenter.clone().add(subCoronalVentNorm.multiplyScalar(ventralSurfaceOffset * 1.04));


      const glansVentralCleftPt = p2.clone()

        .add(shaftEndTan.clone().multiplyScalar(glansTipDist * 0.48))

        .add(endVentNorm.clone().multiplyScalar(0.24 * glansScale));


      const glansApexMeatus = p2.clone()

        .add(shaftEndTan.clone().multiplyScalar(glansTipDist))

        .add(endVentNorm.clone().multiplyScalar(-0.04 * glansScale));


      externalCoursePts.push(subCoronalSurfacePt, glansVentralCleftPt, glansApexMeatus);

      const externalVentralCurve = new THREE.CatmullRomCurve3(externalCoursePts, false, 'centripetal', 0.5);


      // (A) Formed Tubular Urethra (from Bladder Neck up to current External Urethral Meatus `m`)

      const masterTubularPts = [

        bladderNeckPt,

        prostaticUrethraPt,

        membranousUrethraPt

      ];


      const EXT_SAMPLES = 36;

      for (let i = 0; i <= EXT_SAMPLES; i++) {

        const u = (i / EXT_SAMPLES) * m;

        if (m < 0.005 && i > 0) break;

        masterTubularPts.push(externalVentralCurve.getPointAt(u));

      }


      const tubularUrethraCurve = new THREE.CatmullRomCurve3(masterTubularPts, false, 'centripetal', 0.5);

      const urethraRadius = THREE.MathUtils.lerp(0.056, 0.092, g);

      tubularUrethraMesh.geometry.dispose();

      tubularUrethraMesh.geometry = new THREE.TubeGeometry(tubularUrethraCurve, 56, urethraRadius, 18, false);


      // (B) External Urethral Meatus Placement

      const meatusCurvePos = externalVentralCurve.getPointAt(m);

      const meatusTan = externalVentralCurve.getTangentAt(m).normalize();

      const ventNormAtMeatus = new THREE.Vector3(0, -meatusTan.z, meatusTan.y).normalize();


      const blendFactor = THREE.MathUtils.smoothstep(m, 0.86, 0.99);

      const openingDir = ventNormAtMeatus.clone().lerp(meatusTan, blendFactor).normalize();


      const surfaceMeatusPos = meatusCurvePos.clone().add(openingDir.clone().multiplyScalar(urethraRadius * 0.72));

      meatusGroup.position.copy(surfaceMeatusPos);

      meatusGroup.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, 1), openingDir);

      const meatusScale = THREE.MathUtils.lerp(0.85, 1.18, g);

      meatusGroup.scale.setScalar(meatusScale);


      // (C) Non-Overlapping Open Urethral Plate (Distal to Meatus Rim)

      leftPlateRidgeMesh.geometry.dispose();

      rightPlateRidgeMesh.geometry.dispose();

      centerPlateGrooveMesh.geometry.dispose();


      const meatusClearanceU = 0.035;

      const plateStartU = m + meatusClearanceU;


      if (plateStartU < 0.985) {

        const leftRidgePts = [];

        const rightRidgePts = [];

        const centerGroovePts = [];

        const PLATE_SAMPLES = 24;

        const plateHalfWidth = urethraRadius * 0.50;


        for (let i = 0; i <= PLATE_SAMPLES; i++) {

          const u = plateStartU + (1.0 - plateStartU) * (i / PLATE_SAMPLES);

          const pt = externalVentralCurve.getPointAt(u);

          const tan = externalVentralCurve.getTangentAt(u).normalize();

          const vNorm = new THREE.Vector3(0, -tan.z, tan.y).normalize();


          const groovePt = pt.clone().add(vNorm.clone().multiplyScalar(urethraRadius * 0.18));

          const ridgeSurfacePt = pt.clone().add(vNorm.clone().multiplyScalar(urethraRadius * 0.36));


          centerGroovePts.push(groovePt);

          leftRidgePts.push(new THREE.Vector3(-plateHalfWidth, ridgeSurfacePt.y, ridgeSurfacePt.z));

          rightRidgePts.push(new THREE.Vector3(plateHalfWidth, ridgeSurfacePt.y, ridgeSurfacePt.z));

        }


        const ridgeRadius = urethraRadius * 0.32;

        leftPlateRidgeMesh.geometry = new THREE.TubeGeometry(

          new THREE.CatmullRomCurve3(leftRidgePts, false, 'centripetal', 0.5),

          24,

          ridgeRadius,

          12,

          false

        );

        rightPlateRidgeMesh.geometry = new THREE.TubeGeometry(

          new THREE.CatmullRomCurve3(rightRidgePts, false, 'centripetal', 0.5),

          24,

          ridgeRadius,

          12,

          false

        );

        centerPlateGrooveMesh.geometry = new THREE.TubeGeometry(

          new THREE.CatmullRomCurve3(centerGroovePts, false, 'centripetal', 0.5),

          24,

          ridgeRadius * 1.1,

          12,

          false

        );


        leftPlateRidgeMesh.visible = true;

        rightPlateRidgeMesh.visible = !sagittalMode;

        centerPlateGrooveMesh.visible = true;

      } else {

        leftPlateRidgeMesh.visible = false;

        rightPlateRidgeMesh.visible = false;

        centerPlateGrooveMesh.visible = false;

      }

    }

    function setCameraView(view) {

      currentViewName = view;

      const aspect = container.clientWidth / Math.max(container.clientHeight, 1);

      const distScale = aspect < 0.85 ? 1.25 : (aspect < 1.1 ? 1.12 : 1.0);

      const yTargetOffset = aspect < 0.9 ? 0.22 : 0.0;


      if (view === 'oblique') {

        camera.position.set(5.4 * distScale, 2.0 * distScale, 6.4 * distScale);

        controls.target.set(0, 0.15 + yTargetOffset, 0.25);

        camera.position.set(7.2 * distScale, 0.35, 0.1);

        controls.target.set(0, 0.20 + yTargetOffset, 0.1);

      } else if (view === 'ventral') {

        camera.position.set(0, -4.8 * distScale, 4.4 * distScale);

        controls.target.set(0, 0.0 + yTargetOffset * 0.5, 0.35);

      }

      controls.update();

    }

    function animate() {

      requestAnimationFrame(animate);

      let needsUpdate = false;

      ['glans', 'curvature', 'scrotum', 'meatus', 'gonad'].forEach((key) => {

        const diff = targetState[key] - state[key];

        if (Math.abs(diff) > 0.04) {

          state[key] += diff * 0.14;

          needsUpdate = true;

        } else if (state[key] !== targetState[key]) {

          state[key] = targetState[key];

          needsUpdate = true;

        }

      });


      if (needsUpdate) rebuildMorphology();

      controls.update();

      renderer.render(scene, camera);

    }

    animate();
